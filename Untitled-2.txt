def generate_confetti(count=30):
    """生成随机位置的彩屑 HTML"""
    colors = ['#ff6b6b', '#ffd93d', '#6bcb77', '#4d96ff', '#ff922b', '#cc5de8']
    html_parts = []
    for i in range(count):
        c = random.choice(colors)
        left = random.randint(0, 100)
        delay = round(random.random() * 3, 1)
        dur = round(2.5 + random.random() * 3, 1)
        html_parts.append(
            f'<div class="confetti" style="left:{left}%; background:{c}; '
            f'animation-delay:{delay}s; animation-duration:{dur}s;"></div>'
        )
    return '\n  '.join(html_parts)


def generate_bubbles(count=10):
    """生成随机位置的气泡 HTML"""
    colors = ['#ff6b6b', '#ffd93d', '#6bcb77', '#4d96ff', '#ff922b', '#cc5de8']
    html_parts = []
    for i in range(count):
        c = random.choice(colors)
        left = random.randint(0, 90)
        top = random.randint(0, 90)
        size = random.randint(25, 80)
        delay = round(random.random() * 4, 1)
        html_parts.append(
            f'<div class="bubble" style="left:{left}%; top:{top}%; '
            f'width:{size}px; height:{size}px; background:{c}; '
            f'animation-delay:{delay}s;"></div>'
        )
    return '\n  '.join(html_parts)


class ChildrensDayHandler(http.server.SimpleHTTPRequestHandler):
    """六一祝福页面请求处理"""
    visit_count = 0

    def do_GET(self):
        ChildrensDayHandler.visit_count += 1
        html = HTML_TEMPLATE.format(
            confetti=generate_confetti(30),
            bubbles=generate_bubbles(10),
            random_bless=random.choice(BLESSINGS),
            year=2026,
            visit=ChildrensDayHandler.visit_count,
        )
        self.send_response(200)
        self.send_header("Content-Type", "text/html; charset=utf-8")
        self.send_header("Content-Length", len(html.encode('utf-8')))
        self.end_headers()
        self.wfile.write(html.encode('utf-8'))

    def log_message(self, format, *args):
        """美化控制台日志"""
        emojis = ['🎈', '🎠', '🍭', '🎪', '🎡', '🎢']
        print(f"  {random.choice(emojis)} [{self.log_date_time_string()}] {args[0]}")
